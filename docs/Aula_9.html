estande <- estande %>%

3 Visualização Foi utilizada as funções geom_jittler para visualizar a distribuição dos dados e geom_smooth para mostrar a linha de tendência linear ajustada dos dados.

estande %>% 
  ggplot(aes(trat, nplants))+
  geom_jitter(width = 0.2, alpha = 0.2, color = "darkorange")+
  stat_summary(fun.data = "mean_cl_boot")+
  geom_smooth(method = "lm", se = FALSE)+
  facet_wrap(~exp)

estande %>% 
  ggplot(aes(trat, nplants))+
  geom_jitter(width = 0.2, alpha = 0.2, color = "darkblue")+
  stat_summary(fun.data = "mean_cl_boot")+
  geom_smooth(method = "lm", se = FALSE)
